> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 호출자가 Phase 1 응답의 `meta.file_meta[i]` 6 필드를 다시 입력으로 넘기면 (`stored_files=[StoredFile(...)]`), llm-request 가 file-manager 에서 bytes 만 fetch — bytes 재전송 비용을 회피한다.

**Architecture:** 신규 인자 `stored_files: list[StoredFile]` 추가 (기존 `files=[FileInput]` 와 sibling). `StoredFile` 은 FileInput 의 bytes 만 file_path 로 대체된 형태 (5 메타 + file_path = **6 필드**) — Phase 1 ↔ Phase 2 caller mental model 대칭. 진입점 4 함수가 `resolve_stored_files` 로 file-manager 에서 bytes fetch → caller 의 6 필드 그대로 FileInput 으로 변환 → 기존 파이프라인. 1차 `get_file(file_path)` 실패 시 hashing 3 필드로 `check_cache` fallback. 둘 다 miss 면 `FileResolveError` raise.

"""요청 DTO."""

from __future__ import annotations

from base64 import b64decode, b64encode
from typing import Annotated, Union

from pydantic import BaseModel, BeforeValidator, ConfigDict, PlainSerializer


# JSON 직렬화 시 base64 문자열로 인코딩, 파싱 시 raw bytes로 디코딩.
# Pydantic v2의 `bytes` 타입은 JSON 모드에서 UTF-8 디코드를 시도해 바이너리는 실패하므로 필요.
#
# ⚠️ 의도된 중복: 동일 정의가 sdk/llm_request/models.py 에도 존재.
# SDK ↔ Service 는 독립 파이썬 패키지로 공유 모듈이 없으므로 양쪽에 같은 정의 유지.
# 수정 시 양쪽 **반드시** 동기화 (직렬화 동작이 SDK/Service 계약의 일부).
Base64Field = Annotated[
    bytes,
    BeforeValidator(lambda v: b64decode(v) if isinstance(v, str) else v),
    PlainSerializer(lambda b: b64encode(b).decode(), when_used="json", return_type=str),
]


class FileInput(BaseModel):
    """에이전트가 서버에 넘기는 파일 입력.

    file_path는 에이전트가 세팅하지 않는다. Service가 file-manager 등록 단계에서
    check_cache / upload_file 반환값으로 채운 뒤 MDB `file_meta`에 함께 저장한다.

    resolved_via (Phase 2): 어느 경로로 bytes 가 확보됐는지 추적.
    - "upload": data-mode (files=[..]) — 호출자 bytes 를 file-manager 에 신규 업로드
    - "primary": path-mode (stored_files=[..]) — file_path 1차 fetch 성공
    - "fallback": path-mode — file_path stale → check_cache 로 새 path 발견 후 fetch
    """

    file_name: str
    file_size: int
    file_mtime: Union[str, int]
    mime_type: str
    extension: str
    data: Base64Field
    file_path: str | None = None
    resolved_via: str | None = None


class FileKeyingMeta(BaseModel):
    """/v1/done-check 용 경량 메타. hashing 3필드만 포함."""

    file_name: str
    file_size: int
    file_mtime: Union[str, int]


class StoredFile(BaseModel):
    """이미 file-manager 에 등록된 파일 참조 (Phase 2 입력).

    Phase 1 응답의 `meta.file_meta[i]` 6 필드 중 4 필드 (file_path + hashing 3 필드)
    만 받는다. mime_type/extension 은 file-manager 의 get_file 응답에 포함되어 자동 채워짐.

    extra="forbid": 호출자가 Phase 1 응답 dict 를 그대로 넘기면 mime_type 등에서
    ValidationError. 의도된 strict — 키 오타 / 잘못된 입력 즉시 잡기 위함 (P2-4).
    """

    model_config = ConfigDict(extra="forbid")

    file_path: str
    file_name: str
    file_size: int
    file_mtime: Union[str, int]


class LLMRequestBody(BaseModel):
    """POST /v1/request 요청 바디.

    files: 에이전트가 제공하는 파일 리스트 (meta + bytes). 서버가 내부에서
    file-manager에 등록 후 파이프라인에 그대로 전달.

    stored_files (Phase 2): 이미 file-manager 에 등록된 파일 참조. 서버가
    file-manager 에서 fetch 하여 bytes 로딩 후 파이프라인에 합침.
    """

    llm_input: dict
    files: list[FileInput] | None = None
    stored_files: list[StoredFile] | None = None
    cache_mode: bool = True
    save_mode: bool = True
    ttl_seconds: int | None = None
    database: str | None = None
    context_cache_ttl: int | None = None
    standby_ttl_seconds: int = 600
    debug: bool = False


class DoneCheckBody(BaseModel):
    """POST /v1/done-check 요청 바디."""

    llm_input: dict
    files: list[FileKeyingMeta] | None = None
    database: str | None = None


class FetchBody(BaseModel):
    """POST /v1/fetch 요청 바디."""

    cache_key: str
    database: str | None = None

"""file_resolver.py 단위 테스트 — Phase 2.

resolve_stored_files: stored_files 를 file-manager 에서 fetch → FileInput 변환.

FileManagerHandler 는 외부 (file-manager SDK) 클래스이므로 monkeypatch 로 모킹.

from __future__ import annotations

from unittest.mock import AsyncMock, MagicMock

import pytest

from app.core.errors import FileResolveError
from app.models.requests import StoredFile


@pytest.fixture()
def mock_fm(monkeypatch):
    """FileManagerHandler() 호출 시 동일 mock 인스턴스 반환 (싱글톤 흉내)."""
    fm_instance = MagicMock()
    fm_instance.get_file = AsyncMock()
    fm_instance.check_cache = AsyncMock()

    fm_class = MagicMock(return_value=fm_instance)
    monkeypatch.setattr("app.logic.file_resolver.FileManagerHandler", fm_class)
    return fm_instance


def _ref(file_path: str = "abc.pdf", **overrides) -> StoredFile:
    base = {
        "file_path": file_path,
        "file_name": "x.pdf",
        "file_size": 100,
        "file_mtime": 1700000000,
    }
    base.update(overrides)
    return StoredFile(**base)


def _fm_metadata(file_path: str = "abc.pdf"):
    return {
        "file_name": "x.pdf",
        "file_size": 100,
        "file_mtime": 1700000000,
        "mime_type": "application/pdf",
        "extension": ".pdf",
        "file_path": file_path,
    }


class TestResolveStoredFilesPrimary:
    """1차 get_file 성공 → resolved_via='primary'."""

    async def test_single_ref_primary_success(self, mock_fm):
        from app.logic.file_resolver import resolve_stored_files

        mock_fm.get_file.return_value = (b"\x00\x01\x02", _fm_metadata("abc.pdf"))

        result = await resolve_stored_files([_ref("abc.pdf")])

        assert len(result) == 1
        fi = result[0]
        assert fi.data == b"\x00\x01\x02"
        assert fi.file_path == "abc.pdf"
        assert fi.mime_type == "application/pdf"
        assert fi.resolved_via == "primary"
        mock_fm.check_cache.assert_not_awaited()  # fallback 발동 X


class TestResolveStoredFilesFallback:
    """1차 404 → check_cache hit → 2차 get_file 성공 → resolved_via='fallback'."""

    async def test_stale_path_recovers_via_check_cache(self, mock_fm, caplog):
        from app.logic.file_resolver import resolve_stored_files
        from file_manager.exceptions import FMServiceError

        # 1차 get_file: 404
        # 2차 get_file (new_path): 성공
        new_path = "new-uuid.pdf"
        mock_fm.get_file.side_effect = [
            FMServiceError("not found", status_code=404),
            (b"\x10\x20", _fm_metadata(new_path)),
        ]
        mock_fm.check_cache.return_value = {"hit": True, "file_path": new_path}

        with caplog.at_level("WARNING"):
            result = await resolve_stored_files([_ref("stale.pdf")])

        assert len(result) == 1
        fi = result[0]
        assert fi.file_path == new_path
        assert fi.resolved_via == "fallback"
        assert any("stale file_path" in rec.message for rec in caplog.records)
        assert mock_fm.get_file.await_count == 2
        mock_fm.check_cache.assert_awaited_once()


class TestResolveStoredFilesBothMiss:
    """1차 404 + check_cache miss → FileResolveError raise."""

    async def test_raises_file_resolve_error(self, mock_fm):
        from app.logic.file_resolver import resolve_stored_files
        from file_manager.exceptions import FMServiceError

        mock_fm.get_file.side_effect = FMServiceError("not found", status_code=404)

```python
"""Phase 2 — stored_files 를 file-manager 에서 fetch 하여 FileInput 으로 변환.

설계: docs/18.file-meta-response-exposure.md Phase 2 §5.3
"""

from __future__ import annotations

import logging

from file_manager import FileManagerHandler
from file_manager.exceptions import FMServiceError

from app.core.errors import FileResolveError
from app.models.requests import FileInput, StoredFile

logger = logging.getLogger(__name__)


async def resolve_stored_files(refs: list[StoredFile]) -> list[FileInput]:
    """각 StoredFile 을 file-manager 에서 fetch → bytes 로딩된 FileInput 반환.

    1. fm.get_file(ref.file_path) 1차 시도 → 성공 시 resolved_via="primary"
    2. 404 시 fm.check_cache(hash 3 필드) → 새 file_path → 재 fetch + resolved_via="fallback"
    3. 둘 다 miss → FileResolveError raise
    4. 비-404 에러 (5xx / network) → 그대로 전파
    """
    fm = FileManagerHandler()
    return [await _resolve_one(fm, ref) for ref in refs]


async def _resolve_one(fm: FileManagerHandler, ref: StoredFile) -> FileInput:
    try:
        data, fm_meta = await fm.get_file(ref.file_path)
        return _build_file_input(data, fm_meta, resolved_via="primary")
    except FMServiceError as e:
        if e.status_code != 404:
            raise  # 비-404 (network / 5xx) 그대로 전파

        cache = await fm.check_cache({
            "file_name": ref.file_name,
            "file_size": ref.file_size,
            "file_mtime": ref.file_mtime,
        })
        if not cache["hit"]:
            raise FileResolveError(
                f"file_path '{ref.file_path}' not found, "
                f"check_cache miss for file_name={ref.file_name}"
            ) from e

        new_path = cache["file_path"]
        logger.warning(
            "[Phase2] stale file_path: input=%s, resolved=%s, file_name=%s",
            ref.file_path, new_path, ref.file_name,
        )
        data, fm_meta = await fm.get_file(new_path)
        return _build_file_input(data, fm_meta, resolved_via="fallback")


def _build_file_input(data: bytes, fm_meta: dict, *, resolved_via: str) -> FileInput:
    return FileInput(
        data=data,
        file_path=fm_meta["file_path"],
        file_name=fm_meta["file_name"],
        file_size=fm_meta["file_size"],
        file_mtime=fm_meta["file_mtime"],
        mime_type=fm_meta["mime_type"],
        extension=fm_meta["extension"],
        resolved_via=resolved_via,
    )
```

async def llm_request_async(
    llm_input: dict,
    *,
    files: list[FileInput] | None = None,
    stored_files: list[StoredFile] | None = None,
    cache_mode: bool = True,
    save_mode: bool = True,
    ttl_seconds: int | None = None,
    database: str | None = None,
    context_cache_ttl: int | None = None,
    standby_ttl_seconds: int = 600,
    debug: bool = False,
) -> tuple[LLMResponse, dict[str, Any]]:
    """단일 LLM 요청 실행 (async). (LLMResponse, meta) 튜플 반환.

    files (FileInput): 호출자 bytes → file-manager 신규 업로드.
    stored_files (StoredFile, Phase 2): 이미 등록된 파일 참조 → file-manager 에서 fetch.
    둘 다 동시 사용 가능 (mixed). 내부에서 합쳐 이후 파이프라인 진입.

    Args:
        standby_ttl_seconds: cache stampede dedup standby TTL + 대기 timeout.
        debug: True면 meta에 route_reason 포함.
    """
    if not (cache_mode or save_mode):
        files = stored_files = None

    # Phase 2 — stored_files 먼저 resolve (실패 가능성 높음 → fail-fast)
    resolved: list[FileInput] = []
    if stored_files:
        resolved = await resolve_stored_files(stored_files)
        # G2: file-manager 는 우리 SUPPORTED_MIME_TYPES 모름 → 재검증
        validate_mime_types(resolved)
        validate_file_sizes(resolved)

    # 기존 data-mode
    if files:
        validate_mime_types(files)
        validate_file_sizes(files)
        await register_files(files)
        for fi in files:
            fi.resolved_via = "upload"  # P2-5

    # G4: 결정적 순서 — files 먼저, stored_files 나중
    combined: list[FileInput] | None = (files or []) + resolved
    combined = combined or None

    prep = prepare(
        llm_input,
        files=combined,
        cache_mode=cache_mode, save_mode=save_mode,
        ttl_seconds=ttl_seconds, database=database,
        context_cache_ttl=context_cache_ttl,
        standby_ttl_seconds=standby_ttl_seconds,
        debug=debug,
    )
    if prep.cached_output:
        return prep.cached_output

    # 이하 기존 코드 그대로 (dedup standby / _do_llm_call)
    if prep.dc is None:
        return await _do_llm_call(prep, database)

    cache_key = prep.dc.cache_key
    acquired = await dedup.try_acquire_standby(
        "agent_llm", cache_key, standby_ttl_seconds,
    )
    if acquired:
        logger.info("[dedup] role=leader key=%s", cache_key[:16])
        try:
            return await _do_llm_call(prep, database)
        finally:
            await dedup.release_standby("agent_llm", cache_key)

    t_wait = time.monotonic()
    cached = await dedup.wait_for_standby_release(
        "agent_llm", cache_key, standby_ttl_seconds,
        find_cache_fn=lambda: _recheck_agent_llm_cache(prep, database),
    )
    wait_ms = int((time.monotonic() - t_wait) * 1000)
    if cached:
        logger.info("[dedup] role=waiter wait_ms=%d leader_completed=true", wait_ms)
        return cached
    logger.warning("[dedup] role=standby_timeout wait_ms=%d", wait_ms)
    return await _do_llm_call(prep, database)

### `resolved_via` 값

| 값 | 의미 |
